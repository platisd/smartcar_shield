33unsigned int SRF08::getDistance()
34{
35 attach();
36
37 static const uint8_t kFirstEchoHighByte = 0x02;
38 static const uint8_t kNumberOfBytesToRequest = 2;
39
40 measureDistance();
41
42 mRuntime.i2cBeginTransmission(mAddress);
43 mRuntime.i2cWrite(kFirstEchoHighByte);
44 mRuntime.i2cEndTransmission();
45
46 mRuntime.i2cRequestFrom(mAddress, kNumberOfBytesToRequest);
47 if (mRuntime.i2cAvailable() == 0)
48 {
49 return static_cast<unsigned int>(-1); // Return a large error-like value
50 }
51 auto high = mRuntime.i2cRead();
52 auto low = mRuntime.i2cRead();
53
54 static constexpr auto kBitsInByte = 8;
55
56 return static_cast<uint16_t>((high << kBitsInByte) + low);
57}
