37void GY50::update()
38{
39 unsigned long currentTime = mRuntime.currentTimeMillis();
40 unsigned long interval = currentTime - mPreviousSample;
41 if (interval <= kSamplingInterval)
42 {
43 return; // Not the time to read yet
44 }
45
46 int drift = kOffset - getAngularVelocity();
47
48 if (getAbsolute(drift) > kGyroThreshold)
49 {
50 float gyroRate = static_cast<float>(drift) * kGyroSensitivity;
51 static constexpr auto kGyroscopeReadingScale = 1000.0F;
52 mAngularDisplacement += gyroRate / (kGyroscopeReadingScale / static_cast<float>(interval));
53 }
54 mPreviousSample = currentTime;
55}

57void GY50::attach()
58{
59 if (mAttached)
60 {
61 return;
62 }
63
64 mRuntime.i2cInit();
65 // Enable z and turn off power down
66 static constexpr uint8_t kControlRegister1 = 0x20;
67 writeL3G4200DRegister(kControlRegister1, 0b00001100); // NOLINT(readability-magic-numbers)
68 // If you'd like to adjust/use the HPF, you can edit the line below to configure CTRL_REG2
69 static constexpr uint8_t kControlRegister2 = 0x21;
70 writeL3G4200DRegister(kControlRegister2, 0b00000000); // NOLINT(readability-magic-numbers)
71 // Configure CTRL_REG3 to generate data ready interrupt on INT2
72 // No interrupts used on INT1, if you'd like to configure INT1
73 // or INT2 otherwise, consult the datasheet
74 static constexpr uint8_t kControlRegister3 = 0x22;
75 writeL3G4200DRegister(kControlRegister3, 0b00001000); // NOLINT(readability-magic-numbers)
76 // CTRL_REG4 controls the full-scale range, among other things
77 static constexpr uint8_t kControlRegister4 = 0x23;
78 writeL3G4200DRegister(kControlRegister4, 0b00110000); // NOLINT(readability-magic-numbers)
79 // CTRL_REG5 controls high-pass filtering of outputs, use it if you'd like
80 static constexpr uint8_t kControlRegister5 = 0x24;
81 writeL3G4200DRegister(kControlRegister5, 0b00000000); // NOLINT(readability-magic-numbers)
82
83 mAttached = true;
84}

86int GY50::getOffset(int measurements)
87{
88 if (measurements <= 0)
89 {
90 return kError;
91 }
92
93 long sum = 0;
94 for (auto i = 0; i < measurements; i++)
95 {
96 sum += getAngularVelocity();
97 mRuntime.delayMillis(kMeasurementInterval);
98 }
99
100 return static_cast<int>(sum / measurements);
101}

103int GY50::getAngularVelocity()
104{
105 attach();
106
107 static const uint8_t zAxisFirstByteRegister = 0x2D;
108 static const uint8_t zAxisSecondByteRegister = 0x2C;
109
110 auto firstByte = readL3G4200DRegister(zAxisFirstByteRegister);
111 auto secondByte = readL3G4200DRegister(zAxisSecondByteRegister);
112
113 static constexpr auto kBitsInByte = 8;
114
115 return static_cast<int16_t>((firstByte << kBitsInByte) | secondByte);
116}
