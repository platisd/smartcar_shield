67{
68 // Sort the array using bubble sort
69 for (unsigned int i = 0; i < arraySize - 1; i++)
70 {
71 for (unsigned int j = 0; j < arraySize - 1 - i; j++)
72 {
73 if (unsortedNumbers[j] > unsortedNumbers[j + 1])

76 unsortedNumbers[j] = unsortedNumbers[j + 1];
